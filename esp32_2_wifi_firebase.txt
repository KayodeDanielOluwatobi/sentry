// ============================================================
// ACS758 CURRENT SENSOR CALIBRATION & READING
// ============================================================
float readChannelCurrent(LoadChannel& ch) {
    long sum = 0;
    for (int i = 0; i < 50; i++) {
        sum += analogRead(ch.sensorPin);
        delayMicroseconds(100);
    }
    float avgAdc = sum / 50.0;
    float voltage = (avgAdc / 4095.0) * 3.3;
    
    // ACS758-050B sensitivity at 3.3V supply is approx 0.0264 V/A
    float current = (voltage - ch.zeroVoltage) / 0.0264;
    
    // Deadband for sensor noise
    if (abs(current) < 0.3) current = 0.0;
    
    return abs(current);
}

void calibrateSensors() {
    Serial.println("[Sensors] Auto-calibrating zero-points... (Ensure no load current is flowing)");
    for (int i = 0; i < NUM_CHANNELS; i++) {
        long sum = 0;
        int samples = 500;
        for (int j = 0; j < samples; j++) {
            sum += analogRead(channels[i].sensorPin);
            delayMicroseconds(500);
        }
        float avgAdc = sum / (float)samples;
        channels[i].zeroVoltage = (avgAdc / 4095.0) * 3.3;
        
        Serial.printf("[Sensors] %s zero-point calibrated to: %.3f V\n", 
                      channels[i].name, channels[i].zeroVoltage);
    }
    Serial.println("[Sensors] Calibration complete.");
}

// ============================================================
// WIFI & NTP INITIALIZATION
// ============================================================
void connectWiFi() {
    if (WiFi.status() == WL_CONNECTED) return;

    Serial.print("[WiFi] Connecting to: ");
    Serial.println(WIFI_SSID);

    WiFi.mode(WIFI_STA);
    WiFi.begin(WIFI_SSID, WIFI_PASSWORD);

    unsigned long start = millis();
    while (WiFi.status() != WL_CONNECTED && millis() - start < WIFI_TIMEOUT) {
        delay(500);
        Serial.print(".");
    }

    if (WiFi.status() == WL_CONNECTED) {
        Serial.println("\n[WiFi] Connected successfully!");
        Serial.print("[WiFi] IP Address: ");
        Serial.println(WiFi.localIP());

        configTime(GMT_OFFSET, DST_OFFSET, NTP_SERVER1, NTP_SERVER2);
        Serial.println("[NTP] Syncing network time...");
        struct tm timeinfo;
        if (getLocalTime(&timeinfo, 5000)) {
            Serial.println("[NTP] Time synchronized successfully.");
            ntpSynced = true;
        } else {
            Serial.println("[NTP] Failed to sync time on boot.");
        }
    } else {
        Serial.println("\n[WiFi] Connection failed. Will retry in loop.");
    }
}

String chargeState() {
    if (bms.current > 0.5)  return "CHARGING";
    if (bms.current < -0.5) return "DISCHARGING";
    return "IDLE";
}

// ============================================================
// CLOUD PUBLISHING: SUPABASE HISTORICAL DATABASE
// ============================================================
void sendToSupabase() {
    if (WiFi.status() != WL_CONNECTED) return;
    if (!bms.valid) {
        Serial.println("[Supabase] Awaiting valid BMS telemetry before logging historical snapshot...");
        return;
    }

    WiFiClientSecure client;
    client.setInsecure(); // Bypass TLS cert check on ESP32

    HTTPClient http;
    String url = String(SUPABASE_HOST) + SUPABASE_TABLE;
    http.begin(client, url);
    http.addHeader("Content-Type", "application/json");
    http.addHeader("apikey", SUPABASE_API_KEY);
    http.addHeader("Authorization", String("Bearer ") + SUPABASE_API_KEY);
    http.addHeader("Prefer", "return=minimal");

    char tsBuf[32];
    time_t nowSec = time(NULL);
    if (nowSec > 100000) {
        uint64_t ms = ((uint64_t)nowSec) * 1000ULL + (millis() % 1000);
        snprintf(tsBuf, sizeof(tsBuf), "%llu", ms);
    } else {
        uint64_t ms = 1784721310000ULL + millis();
        snprintf(tsBuf, sizeof(tsBuf), "%llu", ms);
    }

    char tempsBuf[64];
    snprintf(tempsBuf, sizeof(tempsBuf), "[%.1f,%.1f,%.1f]",
             bms.temp1, bms.temp2, bms.mosTemp);

    String cellsBuf = "[";
    for (int i = 0; i < bms.activeCells; i++) {
        cellsBuf += String(bms.cellVoltages[i], 3);
        if (i < bms.activeCells - 1) cellsBuf += ",";
    }
    cellsBuf += "]";

    String payload = "{";
    payload += "\"timestamp\":"     + String(tsBuf)              + ",";
    payload += "\"voltage\":"       + String(bms.voltage, 2)     + ",";
    payload += "\"current\":"       + String(bms.current, 2)     + ",";
    payload += "\"power\":"         + String(bms.power, 2)       + ",";
    payload += "\"temperatures\":"  + String(tempsBuf)           + ",";
    payload += "\"cell_voltages\":" + cellsBuf                   + ",";
    payload += "\"is_offline\":false";
    payload += "}";

    int code = http.POST(payload);
    if (code == 201 || code == 200 || code == 204) {
        Serial.println("[Supabase] Telemetry snapshot logged successfully (201 Created).");
    } else {
        Serial.printf("[Supabase] Insert failed: HTTP %d | Response: %s\n", code, http.getString().c_str());
    }
    http.end();
}

// ============================================================
// CLOUD COMMAND POLLING: FIREBASE /commands & /loadManager
// ============================================================
void pollCommands() {
    if (WiFi.status() != WL_CONNECTED) return;

    // 1. Poll /commands.json (Inverter Charge/Discharge control)
    HTTPClient http;
    http.begin(String(FIREBASE_HOST) + "/commands.json");
    int code = http.GET();

    if (code == 200) {
        String body = http.getString();
        StaticJsonDocument<256> doc;
        if (!deserializeJson(doc, body)) {
            bool discharge = doc["discharge"] | true;
            bool charge    = doc["charge"]    | true;

            if (!commandsInitialized) {
                lastKnownDischarge  = discharge;
                lastKnownCharge     = charge;
                commandsInitialized = true;
            } else {
                if (discharge != lastKnownDischarge) {
                    lastKnownDischarge = discharge;
                    StaticJsonDocument<128> cmd;
                    cmd["cmd"] = "discharge";
                    cmd["val"] = discharge;
                    serializeJson(cmd, Serial2);
                    Serial2.println();
                }
                if (charge != lastKnownCharge) {
                    lastKnownCharge = charge;
                    StaticJsonDocument<128> cmd;
                    cmd["cmd"] = "charge";
                    cmd["val"] = charge;
                    serializeJson(cmd, Serial2);
                    Serial2.println();
                }
            }
        }
    }
    http.end();

    // 2. Poll /loadManager.json (Remote Web Relay Overrides)
    HTTPClient httpLM;
    httpLM.begin(String(FIREBASE_HOST) + "/loadManager.json");
    int lmCode = httpLM.GET();

    if (lmCode == 200) {
        String body = httpLM.getString();
        StaticJsonDocument<256> doc;
        if (!deserializeJson(doc, body)) {
            String mode = doc["mode"] | "AUTO";
            bool l1     = doc["channel1"] | true;
            bool l2     = doc["channel2"] | true;
            bool l3     = doc["channel3"] | true;

            if (!loadManagerInitialized) {
                lastKnownMode = mode;
                lastKnownLoad1 = l1;
                lastKnownLoad2 = l2;
                lastKnownLoad3 = l3;
                loadManagerInitialized = true;
            } else {
                // If the remote user clicked anything on the Web Dashboard:
                if (mode != lastKnownMode || l1 != lastKnownLoad1 || l2 != lastKnownLoad2 || l3 != lastKnownLoad3) {
                    lastKnownMode = mode;
                    lastKnownLoad1 = l1;
                    lastKnownLoad2 = l2;
                    lastKnownLoad3 = l3;

                    Serial.printf("[Cloud Web Command] Mode: %s, CH1: %d, CH2: %d, CH3: %d -> Forwarding to Board 1\n",
                                  mode.c_str(), l1, l2, l3);

                    StaticJsonDocument<256> fwd;
                    fwd["cmd"] = "loadManager";
                    fwd["mode"] = mode;
                    fwd["ch1"] = l1;
                    fwd["ch2"] = l2;
                    fwd["ch3"] = l3;

                    serializeJson(fwd, Serial2);
                    Serial2.println();
                }
            }
        }
    }
    httpLM.end();
}

// ============================================================
// PARSE INCOMING UART TELEMETRY FROM BOARD 1
// ============================================================
void processIncomingTelemetry() {
    while (Serial2.available()) {
        String line = Serial2.readStringUntil('\n');
        line.trim();
        if (line.length() < 10) continue;

        StaticJsonDocument<2048> doc;
        DeserializationError err = deserializeJson(doc, line);
        if (err) continue;

        bms.soc          = doc["soc"]  | 0;
        bms.soh          = doc["soh"]  | 0;
        bms.voltage      = doc["v"]    | 0.0f;
        bms.current      = doc["i"]    | 0.0f;
        bms.power        = doc["p"]    | 0.0f;
        bms.capRemaining = doc["cap"]  | 0.0f;
        bms.fullCap      = doc["fcap"] | 0.0f;
        bms.temp1        = doc["t1"]   | 0.0f;
        bms.temp2        = doc["t2"]   | 0.0f;
        bms.mosTemp      = doc["mos"]  | 0.0f;
        bms.balCurrent   = doc["bal"]  | 0.0f;
        bms.cycleCount   = doc["cyc"]  | 0;
        bms.runtime      = doc["rt"]   | 0;
        bms.charging     = doc["cm"]   | false;
        bms.discharging  = doc["dm"]   | false;
        bms.balancing    = doc["blk"]  | false;
        bms.heating      = doc["htg"]  | false;
        bms.errorBits    = doc["err"]  | 0;
        bms.chargeStatus = doc["cs"]   | 0;
        bms.rssi         = doc["rssi"] | 0;
        bms.activeCells  = doc["cells"]| 0;
        bms.deltaVoltage = doc["dv"]   | 0.0f;
        bms.avgVoltage   = doc["avg"]  | 0.0f;

        // Authoritative Relay states from Board 1
        String m = doc["mode"] | "AUTO";
        isAutoMode = (m == "AUTO");
        channels[0].isOn = (doc["l1"] | 1) == 1;
        channels[1].isOn = (doc["l2"] | 1) == 1;
        channels[2].isOn = (doc["l3"] | 1) == 1;

        JsonArray cv = doc["cv"];
        for (int i = 0; i < bms.activeCells && i < 32; i++) {
            bms.cellVoltages[i] = cv[i] | 0.0f;
        }

        JsonArray cr = doc["cr"];
        for (int i = 0; i < bms.activeCells && i < 32; i++) {
            bms.cellResistances[i] = cr[i] | 0.0f;
        }

        bms.valid = true;
        Serial.printf("[UART-RX] Telemetry Received: SOC=%d%%, V=%.2fV, Mode=%s, Relays=[%d,%d,%d]\n",
                      bms.soc, bms.voltage, isAutoMode ? "AUTO" : "MANUAL",
                      channels[0].isOn, channels[1].isOn, channels[2].isOn);

        // Upload instantly to Firebase
        sendToFirebase();
    }
}

// ============================================================
// SETUP
// ============================================================
void setup() {
    Serial.begin(115200);
    Serial2.setRxBufferSize(2048);
    Serial2.begin(115200, SERIAL_8N1, UART_RX_PIN, UART_TX_PIN);
    Serial2.setTimeout(50);
    delay(1000);

    Serial.println("ESP32 #2 — Cloud Gateway + Current Sensing");
    Serial.println("UART RX  : GPIO16 <- ESP32 #1 TX (GPIO 17)");
    Serial.println("UART TX  : GPIO17 -> ESP32 #1 RX (GPIO 16)");
    Serial.println("Sensors  : CH1:GPIO32 | CH2:GPIO33 | CH3:GPIO34");
    Serial.println("==========================================");

    calibrateSensors();
    connectWiFi();
}

// ============================================================
// MAIN LOOP
// ============================================================
void loop() {
    if (WiFi.status() != WL_CONNECTED) {
        connectWiFi();
    }

    // Process incoming BMS Telemetry and Relay States from Board 1
    processIncomingTelemetry();

    unsigned long now = millis();

    // Poll Firebase for remote user commands
    if (now - lastCommandPoll >= COMMAND_POLL_INTERVAL_MS) {
        lastCommandPoll = now;
        pollCommands();
    }

    // Upload to Supabase History every 5 seconds
    if (now - lastSupabaseUpload >= SUPABASE_INTERVAL_MS) {
        lastSupabaseUpload = now;
        sendToSupabase();
    }

    // Send Sensor Readings & Heartbeat to Board 1 every 3 seconds
    if (now - lastStatusSent >= STATUS_INTERVAL_MS) {
        lastStatusSent = now;

        // Read current sensors
        for (int i = 0; i < NUM_CHANNELS; i++) {
            channels[i].current = channels[i].isOn ? readChannelCurrent(channels[i]) : 0.0;
        }

        StaticJsonDocument<256> stat;
        stat["cmd"]  = "status";
        stat["wifi"] = (WiFi.status() == WL_CONNECTED);
        stat["c1"]   = serialized(String(channels[0].current, 1));
        stat["c2"]   = serialized(String(channels[1].current, 1));
        stat["c3"]   = serialized(String(channels[2].current, 1));

        struct tm timeinfo;
        if (getLocalTime(&timeinfo, 10)) {
            char timeStr[10];
            strftime(timeStr, sizeof(timeStr), "%H:%M", &timeinfo);
            stat["time"] = timeStr;
        } else {
            stat["time"] = "--:--";
        }

        serializeJson(stat, Serial2);
        Serial2.println();
    }

    delay(50);
}
